		pedidoValido(id);
		
		this.id = id;
		
		//if (nombreProducto == null) throw new NullPointerException();
		
		this.nombreProducto = 
			Objects.requireNonNull(nombreProducto, "El producto no debe ser null").trim();
		if (nombreProducto.trim().isEmpty())

			throw new IllegalArgumentException("La cantidad debe ser mayor que 0");

			throw new Exception("[" + id + "] Formato debe ser: P-000");
	}

		return "Pedido [id=" + id + ", nombreProducto=" + nombreProducto + ", descripcionProducto="
				+ descripcionProducto + ", cantidadProducto=" + cantidadProducto + "]";
	}
}

import java.util.Objects;

class Pedido {
	private String id; // P-000
	private String nombreProducto;
	private String descripcionProducto;
	private int cantidadProducto;

	public Pedido(String id, String nombreProducto, String descripcionProducto, int cantidadProducto) throws Exception {
		this.id = validarId(id);
		this.nombreProducto = validarNombre(nombreProducto);
		this.descripcionProducto = validarDescripcion(descripcionProducto);
		this.cantidadProducto = validarCantidad(cantidadProducto);
	}

	private String validarId(String id) {
		Objects.requireNonNull(id, "El id no puede ser null");
		
		if (!id.matches("P-\\d{3}")) {
			throw new IllegalArgumentException("El id debe tener formato P-000");
		}
		return id;
	}

	private String validarNombre(String nombre) {
		Objects.requireNonNull(nombre, "El nombre no puede ser null");
		nombre = nombre.trim();
		if (nombre.isEmpty()) {
			throw new IllegalArgumentException("El nombre no puede estar vacío");
		}
		return nombre;
	}

	private String validarDescripcion(String descripcion) {
		return (descripcion == null) ? "" : descripcion.trim();
	}

	private int validarCantidad(int cantidad) {
		if (cantidad <= 0) {
			throw new IllegalArgumentException("La cantidad debe ser mayor que cero");
		}
		return cantidad;
	}

		if (!id.matches("P-\\d{3}"))
			throw new Exception("[" + id + "] Formato debe ser: P-000");
	}

	@Override
	public String toString() {
		return "Pedido [id=" + id + ", nombreProducto=" + nombreProducto + ", descripcionProducto="
				+ descripcionProducto + ", cantidadProducto=" + cantidadProducto + "]";

}
